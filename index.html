    CHAT SERVER

    Chat between devices runs on a Supabase project. You only need to change this to use your own server.

    1: RUN IN THE SUPABASE SQL EDITOR 
    -- APEXCOACH chat: private rooms, no direct table access for the public key
create table if not exists apex_messages (
  id bigint generated by default as identity primary key,
  room text not null check (char_length(room) between 8 and 64),
  sender text not null check (sender in ('coach', 'client')),
  sender_name text check (char_length(sender_name) <= 60),
  body text not null check (char_length(body) between 1 and 2000),
  plan_id text,
  created_at timestamptz not null default now()
);
create index if not exists apex_messages_room_idx on apex_messages (room, created_at);
alter table apex_messages enable row level security;

create or replace function apex_send(p_room text, p_sender text, p_sender_name text, p_body text, p_plan_id text default null)
returns apex_messages language sql security definer set search_path = public as $$
  insert into apex_messages (room, sender, sender_name, body, plan_id)
  values (p_room, p_sender, p_sender_name, p_body, p_plan_id)
  returning *;
$$;

create or replace function apex_history(p_room text)
returns setof apex_messages language sql security definer set search_path = public as $$
  select * from (
    select * from apex_messages where room = p_room order by created_at desc limit 200
  ) recent order by created_at asc;
$$;

revoke all on apex_messages from anon, authenticated;
grant execute on function apex_send(text, text, text, text, text) to anon, authenticated;
grant execute on function apex_history(text) to anon, authenticated;
